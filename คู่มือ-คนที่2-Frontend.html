คู่มือทีม 3 คน: ระบบประเมินบุคลากร (รายการที่ 57) รันด้วย Docker

คู่มือนี้พาทำระบบตามโค้ด vue-pic-student ตั้งแต่โฟลเดอร์ว่างจนจบ แบ่งงานให้ 3 คน และเรียงลำดับตาม เกณฑ์ให้คะแนน 100 คะแนน ทุกขั้นจบด้วย จุดเช็ก ถ้าผ่านให้ บันทึก (git commit) ทันที ถ้าขั้นถัดไปพัง ก็ย้อนกลับมาจุดที่ผ่านได้ และระบบยังเปิดโชว์กรรมการได้ คะแนนส่วนก่อนหน้าไม่หาย

-- 8 ตาราง อ่านจากบนลงล่าง ตารางลูกอ้างถึงตารางแม่ด้วย FOREIGN KEY
CREATE TABLE IF NOT EXISTS users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  username VARCHAR(50) NOT NULL UNIQUE,
  password VARCHAR(255) NOT NULL,
  full_name VARCHAR(100) NOT NULL,
  department VARCHAR(100) NOT NULL DEFAULT '',
  position VARCHAR(100) NOT NULL DEFAULT '',
  role ENUM('admin','evaluatee','evaluator') NOT NULL,
  active BOOLEAN NOT NULL DEFAULT TRUE,
  must_change_password BOOLEAN NOT NULL DEFAULT TRUE,
  token_version INT NOT NULL DEFAULT 0
);
CREATE TABLE IF NOT EXISTS periods (
  id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  year INT NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL,
  status ENUM('draft','open','closed') NOT NULL DEFAULT 'draft',
  published BOOLEAN NOT NULL DEFAULT FALSE
);
CREATE TABLE IF NOT EXISTS topics (
  id INT AUTO_INCREMENT PRIMARY KEY,
  period_id INT NOT NULL,
  name VARCHAR(100) NOT NULL,
  FOREIGN KEY (period_id) REFERENCES periods(id)
);

  id INT AUTO_INCREMENT PRIMARY KEY,
  topic_id INT NOT NULL,

  weight DECIMAL(8,2) NOT NULL,
  score_type ENUM('scale4','boolean') NOT NULL,
  evidence_type ENUM('none','file','url') NOT NULL,
  evidence_required BOOLEAN NOT NULL DEFAULT FALSE,
  FOREIGN KEY (topic_id) REFERENCES topics(id)
);

  id INT AUTO_INCREMENT PRIMARY KEY,
  period_id INT NOT NULL,
  user_id INT NOT NULL,
  submitted BOOLEAN NOT NULL DEFAULT FALSE,

  FOREIGN KEY (period_id) REFERENCES periods(id),
  FOREIGN KEY (user_id) REFERENCES users(id)
);

เข้าใจโค้ด (คนที่ 1 ต้องอธิบายกรรมการได้)

ตาราง เก็บอะไร สัมพันธ์กับ
users ผู้ใช้ทุกบทบาท (admin, evaluatee ผู้รับ, evaluator กรรมการ) —
periods รอบประเมิน วันเริ่ม–สิ้นสุด สถานะ draft/open/closed —
topics หัวข้อการประเมินในแต่ละรอบ period_id → periods
indicators ตัวชี้วัด น้ำหนัก ชนิดคะแนน (scale4 1–4 หรือ boolean มี/ไม่มี) ชนิดหลักฐาน topic_id → topics
participants ผู้รับที่อยู่ในรอบ period_id → periods, user_id → users
assignments กรรมการที่ถูกมอบหมายให้ผู้รับ หน้าที่ประธาน/กรรมการร่วม ความเห็น ลายเซ็น participant_id → participants, evaluator_id → users
self_scores คะแนนที่ผู้รับประเมินตนเอง + หลักฐาน participant_id, indicator_id
scores คะแนนที่กรรมการให้ assignment_id, indicator_id
  • ชนิดข้อมูล: INT AUTO_INCREMENT รหัส, VARCHAR(n) ข้อความสั้น, TEXT ข้อความยาว, DATE วันที่, DECIMAL(8,2) น้ำหนักทศนิยม, ENUM ค่าที่เลือกได้จำกัด, BOOLEAN ใช่/ไม่ใช่
  • ชื่อตาราง/ฟิลด์เป็นภาษาอังกฤษตัวเล็ก คั่นด้วย _ (snake_case) ตารางเป็นพหูพจน์ คีย์นอกลงท้าย _id
  • UNIQUE (period_id,user_id) กันเพิ่มคนเดิมซ้ำในรอบเดียว

function text(value, max = 100, required = true) {
  const result = String(value ?? '').trim();
  check((!required || result.length > 0) && result.length <= max, 'ข้อมูลว่างหรือยาวเกินกำหนด');
  return result;
}

  res.json({ status: 'success', message, data });
}
function route(fn) {
  return (req, res, next) => Promise.resolve(fn(req, res, next)).catch(next);
}
async function getPeriod(id) {
  const rows = await query('SELECT * FROM periods WHERE id=?', [id]);
  check(rows.length, 'ไม่พบรอบ', 404);
  return rows[0];
}
async function getParticipant(id) {
  const rows = await query(
    'SELECT p.*,u.full_name FROM participants p JOIN users u ON u.id=p.user_id WHERE p.id=?',
    [id],
  );
  check(rows.length, 'ไม่พบผู้รับในรอบ', 404);
  return rows[0];
}
async function items(periodId) {
  return query(
    'SELECT i.*,t.name AS topic FROM indicators i JOIN topics t ON t.id=i.topic_id WHERE t.period_id=? ORDER BY t.id,i.id',
    [periodId],
  );
}
function draft(period) {
  check(period.status === 'draft', 'แก้ได้เฉพาะรอบฉบับร่าง');
}
function open(period) {
  const today = new Date().toLocaleDateString('sv-SE', { timeZone: 'Asia/Bangkok' });
  check(
    period.status === 'open' && today >= period.start_date && today <= period.end_date,
    'รอบยังไม่เปิด หรืออยู่นอกวันประเมิน',
    403,
  );
}
function scoreValid(item, score) {
  return (
    Number.isInteger(score) &&
    (item.score_type === 'boolean' ? [0, 1].includes(score) : [1, 2, 3, 4].includes(score))
  );
}
function percentage(indicators, scores) {
  if (!indicators.length) return null;
  let sum = 0,
    weight = 0;
  for (const item of indicators) {
    const row = scores.find((s) => s.indicator_id === item.id);
    if (!row || !scoreValid(item, row.score)) return null;
    const max = item.score_type === 'boolean' ? 1 : 4;
    sum += (row.score / max) * Number(item.weight);
    weight += Number(item.weight);
  }
  return Math.round((sum / weight) * 10000) / 100;
}
function complete(item, row) {
  if (!row || !scoreValid(item, row.score) || !row.description.trim()) return false;
  if (!item.evidence_required) return true;
  return item.evidence_type === 'file' ? !!row.evidence_file : !!row.evidence_url;
}

  • text(value, max, required) ตัดช่องว่าง ตรวจว่าไม่ว่างและไม่ยาวเกิน (Validate)

// รันทุกครั้งที่ backend เริ่ม: สร้างตารางจาก schema.sql และบัญชีฝึก ไม่ลบข้อมูลเดิม
const fs = require('node:fs');
const path = require('node:path');
const mysql = require('mysql2/promise');
const bcrypt = require('bcryptjs');
const { settings } = require('./db');
async function seed() {
  const db = await mysql.createConnection({ ...settings, multipleStatements: true });
  try {
    await db.query(fs.readFileSync(path.join(__dirname, 'schema.sql'), 'utf8'));
    const users = [
      ['admin', 'Admin1234!', 'ฝ่ายบุคลากร', 'admin'],
      ['emp001', 'Emp1234!', 'ผู้รับหนึ่ง', 'evaluatee'],
      ['emp002', 'Emp1234!', 'ผู้รับสอง', 'evaluatee'],
      ['eval01', 'Eval1234!', 'กรรมการหนึ่ง', 'evaluator'],
      ['eval02', 'Eval1234!', 'กรรมการสอง', 'evaluator'],
    ];
    for (const [username, password, name, role] of users) {
      await db.execute(
        'INSERT IGNORE INTO users(username,password,full_name,role) VALUES(?,?,?,?)',
        [username, await bcrypt.hash(password, 10), name, role],
      );
    }
    console.log('seed เสร็จ ฐานข้อมูล:', settings.database);
  } finally {
    await db.end();
  }
}
seed().catch((error) => {
  console.error('seed ไม่สำเร็จ:', error.code || error.message);
  process.exit(1);
});
  • รันทุกครั้งที่ backend เริ่ม แต่ใช้ CREATE TABLE IF NOT EXISTS และ INSERT IGNORE จึงไม่ลบข้อมูลเดิม
  • รหัสผ่านเก็บแบบ hash ด้วย bcrypt ไม่เก็บรหัสจริง

export async function api(url, method = 'GET', data) {
  const headers = { Authorization: 'Bearer ' + (localStorage.getItem('split_token') || '') };
  const options = { method, headers };
  if (data instanceof FormData) options.body = data;
  else if (data !== undefined) {
    headers['Content-Type'] = 'application/json';
    options.body = JSON.stringify(data);
  }

export async function fileBlob(url) {
  const response = await fetch('/api' + url, {
    headers: { Authorization: 'Bearer ' + localStorage.getItem('split_token') },
  });
  if (!response.ok) {
    const data = await response.json();
    throw Error(data.message);
  }

}

  a.href = blob;

}

      await this.load();

      await api(

<script>
import { api, mayEdit, openFile } from '../api';
import ScoreField from '../components/ScoreField.vue';
export default {
  components: { ScoreField },
  data() {
    return { list: [], data: null, rows: {}, saved: {}, files: {} };
  },
  async mounted() {
    this.list = await api('/my-periods');
  },
  computed: {
    editable() {
      return this.data && !this.data.person.submitted && mayEdit(this.data.period);
    },
    dirty() {
      return Object.keys(this.rows).some((id) => JSON.stringify(this.rows[id]) !== this.saved[id]);
    },
    completed() {
      return this.data ? this.data.indicators.filter((i) => this.ready(i)).length : 0;
    },
  },
  methods: {
    async select(id) {
      if (this.dirty && !confirm('มีข้อมูลยังไม่บันทึก ต้องการเปลี่ยนรอบหรือไม่?')) return;
      this.data = await api('/self/' + id);
      this.rows = {};
      this.saved = {};
      this.files = {};
      for (const i of this.data.indicators) {
        const old = this.data.scores.find((s) => s.indicator_id === i.id);
        this.rows[i.id] = {
          score: old ? old.score : null,
          description: old ? old.description : '',
          evidence_url: old ? old.evidence_url : '',
        };
        this.saved[i.id] = JSON.stringify(this.rows[i.id]);
      }
    },
    ready(i) {
      const s = this.data.scores.find((s) => s.indicator_id === i.id);
      return (
        s &&
        s.description.trim() &&
        (!i.evidence_required || (i.evidence_type === 'file' ? s.evidence_file : s.evidence_url))
      );
    },
    savedRow(i) {
      return this.data.scores.find((s) => s.indicator_id === i.id);
    },
    async refresh() {
      this.data = await api('/self/' + this.data.person.id);
    },
    async save(i) {
      await api('/self/' + this.data.person.id + '/scores/' + i.id, 'PUT', this.rows[i.id]);
      this.saved[i.id] = JSON.stringify(this.rows[i.id]);
      await this.refresh();
      this.$root.message = 'บันทึก ' + i.name + ' แล้ว';
    },
    async upload(i) {
      if (!this.files[i.id]) throw Error('เลือกไฟล์ก่อน');
      const form = new FormData();
      form.append('file', Array.isArray(this.files[i.id]) ? this.files[i.id][0] : this.files[i.id]);
      await api('/self/' + this.data.person.id + '/files/' + i.id, 'POST', form);
      await this.refresh();
      this.$root.message = 'แนบไฟล์แล้ว';
    },
    async evidence(i) {
      await openFile('/files/' + this.savedRow(i).id);
    },
    async submit() {
      if (this.dirty) throw Error('บันทึกข้อมูลที่แก้ไขก่อนส่ง');
      if (confirm('ส่งแล้วแก้ไม่ได้ ยืนยันส่ง?')) {
        await api('/self/' + this.data.person.id + '/submit', 'POST');
        await this.refresh();
        this.$root.message = 'ส่งการประเมินตนเองแล้ว';
      }
    },
  },
};
</script>

import { api, mayEdit, fileBlob, openFile } from '../api';
import ScoreField from '../components/ScoreField.vue';
export default {
  components: { ScoreField },
  data() {
    return {
      list: [],
      data: null,
      values: {},
      saved: {},
      comment: '',
      savedComment: '',
      file: null,
      signature: '',
    };
  },
  async mounted() {
    this.list = await api('/my-assignments');
  },
  beforeUnmount() {
    if (this.signature) URL.revokeObjectURL(this.signature);
  },
  computed: {
    editable() {
      return (
        this.data &&
        !this.data.assignment.submitted &&
        this.data.person.submitted &&
        mayEdit(this.data.period)
      );
    },
    dirty() {
      return (
        this.comment !== this.savedComment ||
        Object.keys(this.values).some((id) => this.values[id] !== this.saved[id])
      );
    },
  },
  methods: {
    async select(id) {
      if (this.dirty && !confirm('มีข้อมูลยังไม่บันทึก เปลี่ยนงานหรือไม่?')) return;
      this.data = await api('/evaluate/' + id);
      this.values = {};
      this.saved = {};
      this.file = null;
      for (const i of this.data.indicators) {
        const s = this.data.scores.find((s) => s.indicator_id === i.id);
        this.values[i.id] = s ? s.score : null;
        this.saved[i.id] = this.values[i.id];
      }
      this.comment = this.data.assignment.comment || '';
      this.savedComment = this.comment;
      await this.showSignature();
    },
    async refresh() {
      this.data = await api('/evaluate/' + this.data.assignment.id);
    },
    self(i) {
      return this.data.self_scores.find((s) => s.indicator_id === i.id) || {};
    },
    async save(i) {
      await api('/evaluate/' + this.data.assignment.id + '/scores/' + i.id, 'PUT', {
        score: this.values[i.id],
      });
      this.saved[i.id] = this.values[i.id];
      await this.refresh();
      this.$root.message = 'บันทึกคะแนนแล้ว';
    },
    async saveComment() {
      await api('/evaluate/' + this.data.assignment.id + '/comment', 'PUT', {
        comment: this.comment,
      });
      this.savedComment = this.comment;
      await this.refresh();
      this.$root.message = 'บันทึกความเห็นแล้ว';
    },
    async showSignature() {
      if (this.signature) URL.revokeObjectURL(this.signature);
      this.signature = this.data.assignment.signature
        ? await fileBlob('/signatures/' + this.data.assignment.id)
        : '';
    },
    async upload() {
      if (!this.file) throw Error('เลือกภาพลายเซ็นก่อน');
      const f = new FormData();
      f.append('file', Array.isArray(this.file) ? this.file[0] : this.file);
      await api('/evaluate/' + this.data.assignment.id + '/signature', 'POST', f);
      await this.refresh();
      await this.showSignature();
      this.$root.message = 'แนบลายเซ็นแล้ว';
    },
    async evidence(i) {
      await openFile('/files/' + this.self(i).id);
    },
    async submit() {

      if (confirm('ส่งแล้วแก้ไม่ได้ ยืนยัน?')) {

    }

    async select(id) {
      this.loading = true;
      try {
        this.result = await api('/results/' + id);
        for (const url of Object.values(this.signatures)) URL.revokeObjectURL(url);
        this.signatures = {};
        for (const c of this.result.committees)
          if (c.signature) this.signatures[c.id] = await fileBlob('/signatures/' + c.id);

        this.loading = false;

      if (this.exporting || this.loading) return;
      this.exporting = true;
      let paper;
      try {
        await document.fonts.ready;
        // คัดลอกรายงานเป็นกระดาษขนาดคงที่ จึงส่งออกได้แม้เปิดบนจอเล็ก

          node.style.overflow = 'visible';

        await html2pdf()
          .set({
            filename: 'evaluation-' + (this.participantId || this.periodId || 'report') + '.pdf',
            margin: 10,
            image: { type: 'jpeg', quality: 0.98 },
            html2canvas: { scale: 2, backgroundColor: '#ffffff' },
            jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' },
            pagebreak: { mode: ['css', 'legacy'], avoid: ['tr', '.signature', 'h2', '.committee'] },

          .from(paper)
          .save();
